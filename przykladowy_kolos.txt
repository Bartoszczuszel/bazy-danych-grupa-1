b) SELECT d.nazwa, SUM(p.zarobki) AS suma_zarobkow, AVG(p.zarobki) AS srednia_zarobkow) FROM departament d JOIN pracownicy p ON d.id_departamentu = p.id_departamentu GROUP BY d.id_departamentu, d.nazwa HAVING COUNT(p.id_pracownikow) >= 3;

c)
SELECT imie, nazwisko, zarobki FROM pracownicy WHERE zarobki > (SELECT AVG(zarobki) FROM pracownicy WHERE id_departamentu = (SELECT id_departamentu FROM departament WHERE nazwa = 'Shipping'));

d)
SELECT d.nazwa FROM departament d LEFT JOIN pracownicy p ON d.id_departamentu = p.id_departamentu WHERE p.id_pracownika IS NULL;

SELECT p.imie, p.nazwisko, p.data_zatrudnienia FROM pracownicy p JOIN department d ON p.id_departamentu = d.id_departamentu WHERE d.nazwa = 'Shipping' ORDER BY p.data_zatrudnienia DESC;
